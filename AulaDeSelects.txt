-- 1. Criação da tabela CITIES (Cidades)
CREATE TABLE CITIES (
    id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL
);

-- 2. Criação da tabela USERS (Usuários)
CREATE TABLE USERS (
    id INT PRIMARY KEY,
    city_id INT NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    age INT,
    FOREIGN KEY (city_id) REFERENCES CITIES(id)
);
